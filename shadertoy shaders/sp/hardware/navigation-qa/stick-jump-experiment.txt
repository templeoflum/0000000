 function jumpArea(x,y){if(edit||modal()){show('Finish editing / close the menu before jumping areas');return;}const ps=panes();if(!ps.length)return;const column=x<-.35?0:x>.35?2:1,row=y>=0?0:1;const rows=[[],[]];const top=Math.min(...ps.map(p=>p.getBoundingClientRect().top)),bottom=Math.max(...ps.map(p=>p.getBoundingClientRect().top));for(const p of ps)rows[p.getBoundingClientRect().top>(top+bottom)/2?1:0].push(p);const group=rows[row].sort((a,b)=>a.getBoundingClientRect().left-b.getBoundingClientRect().left);const target=group[Math.min(column,group.length-1)];if(target){const lastFocus=remembered.get(target);pick(visible(lastFocus)?lastFocus:all(target)[0]);show('JUMP · '+(target.querySelector('.pane-head strong')?.textContent||target.dataset.module)+' · D-pad selects · ✕ acts');}}
